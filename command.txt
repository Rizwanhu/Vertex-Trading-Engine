# Run from: backend folder (venv activated)

# --- API ---
uvicorn main:app --reload --host 0.0.0.0 --port 8000

# --- Celery worker (Windows: MUST use --pool=solo) ---
# STOP old worker first (Ctrl+C), then start fresh — code changes are NOT picked up otherwise
# Upstash rediss://: ssl_cert_reqs is set automatically in celery_app (CERT_NONE)
celery -A app.workers.celery_app worker --loglevel=info --pool=solo -c 1

# --- Celery beat (scheduled tasks) — run in a SECOND terminal ---
celery -A app.workers.celery_app beat --loglevel=info

# --- Frontend ---
cd ..\frontend
npm run dev

# --- Local Redis (optional, if Upstash DNS fails) ---
# From project root: docker compose up redis -d
# Then in backend/.env: REDIS_URL=redis://localhost:6379/0
